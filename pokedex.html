Projeto Pokédex

Aplicação desenvolvida em Python utilizando a API pública da PokéAPI.

Sobre o Projeto

A Pokédex permite buscar informações de Pokémons usando API, visualizar habilidades, tipos e salvar favoritos em JSON.

Tecnologias Utilizadas

Python
API REST
JSON
Requests

Código do Projeto

api.py


import requests
import urllib3

urllib3.disable_warnings(
    urllib3.exceptions.InsecureRequestWarning
)

def buscar_pokemon_api(nome):

    url = f"https://pokeapi.co/api/v2/pokemon/{nome.lower()}"

    resposta = requests.get(url, verify=False)

    if resposta.status_code == 200:

        dados = resposta.json()

        pokemon = {

            "nome": dados["name"].title(),

            "altura": dados["height"],

            "peso": dados["weight"],

            "tipos": [
                tipo["type"]["name"]
                for tipo in dados["types"]
            ],

            "habilidades": [
                habilidade["ability"]["name"]
                for habilidade in dados["abilities"]
            ]
        }

        return pokemon

    return None

main.py


from api import buscar_pokemon_api
from favoritos import salvar_favorito, ver_favoritos
from pokedex import listar_pokemon

while True:

    print("\n===== POKEDEX =====")
    print("1 - Buscar Pokémon")
    print("2 - Ver Favoritos")
    print("3 - Sair")

    opcao = input("Escolha uma opção: ")

    if opcao == "1":

        nome = input("\nDigite o nome do Pokémon: ")

        pokemon = buscar_pokemon_api(nome)

        if pokemon:

            listar_pokemon(pokemon)

            salvar = input(
                "\nDeseja salvar nos favoritos? (S/N): "
            )

            if salvar.lower() == "s":

                salvar_favorito(pokemon)

    elif opcao == "2":

        ver_favoritos()

    elif opcao == "3":

        print("\nEncerrando Pokédex...")
        break

    else:

        print("\nOpção inválida!")
